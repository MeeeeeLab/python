```python
import csv
import io


class StudentRecord:
    def __init__(self, student_id, name, score):
        student_id, name = student_id.strip(), name.strip()
        if not student_id or not name:
            raise ValueError("id and name are required")
        score = int(score)
        if not 0 <= score <= 100:
            raise ValueError("invalid score")
        self.student_id, self.name, self.score = student_id, name, score

    def to_row(self):
        return {"student_id": self.student_id, "name": self.name, "score": self.score}


class GradeBook:
    def __init__(self):
        self._records = {}

    def add(self, record):
        if record.student_id in self._records:
            return False
        self._records[record.student_id] = record
        return True

    def to_csv(self):
        stream = io.StringIO(newline="")
        writer = csv.DictWriter(stream, fieldnames=["student_id", "name", "score"])
        writer.writeheader()
        writer.writerows(record.to_row() for record in self._records.values())
        return stream.getvalue()

    @classmethod
    def from_csv(cls, text):
        restored = cls()
        reader = csv.DictReader(io.StringIO(text))
        if reader.fieldnames != ["student_id", "name", "score"]:
            raise ValueError("invalid CSV header")
        for row in reader:
            record = StudentRecord(row["student_id"], row["name"], row["score"])
            if not restored.add(record):
                raise ValueError("duplicate student id")
        return restored

    def snapshot(self):
        return [(r.student_id, r.name, r.score) for r in self._records.values()]

    def report(self):
        return "\n".join(
            f"{r.student_id} | {r.name} | {r.score}"
            for r in sorted(self._records.values(), key=lambda item: item.student_id)
        )


def run_checks():
    empty = GradeBook.from_csv(GradeBook().to_csv())
    assert empty.snapshot() == []
    book = GradeBook()
    assert book.add(StudentRecord("S01", "林安", 95))
    assert book.add(StudentRecord("S02", "Bo", 0))
    assert not book.add(StudentRecord("S01", "Other", 60))
    restored = GradeBook.from_csv(book.to_csv())
    assert restored.snapshot() == [("S01", "林安", 95), ("S02", "Bo", 0)]
    assert restored.report() == "S01 | 林安 | 95\nS02 | Bo | 0"
    try:
        GradeBook.from_csv("student_id,name,score\nS03,Cat,101\n")
        raise AssertionError("invalid backup accepted")
    except ValueError:
        pass
    try:
        GradeBook.from_csv("student_id,name,score\nS01,Amy,90\nS01,Bo,80\n")
        raise AssertionError("duplicate id accepted")
    except ValueError:
        pass
    try:
        GradeBook.from_csv("id,name,score\nS01,Amy,90\n")
        raise AssertionError("invalid header accepted")
    except ValueError:
        pass


if __name__ == "__main__":
    run_checks()
    print("gradebook-backup Sunday checks: OK")
```
